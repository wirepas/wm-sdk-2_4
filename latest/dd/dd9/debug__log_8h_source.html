206#define LOG(level, fmt, ...) \
207{ \
208 ((uint8_t) level) <= ((uint8_t) DEBUG_LOG_MAX_LEVEL) ? \
209 Print_Log( \
210 S_MOD_NAME_PREFIX S_TIME_PREFIX S_LEVEL_PREFIX(level) \
211 S_FUNCTION_PREFIX S_LINE_PREFIX \
212 fmt S_LINEENDING S_TIME_SUFFIX S_FUNCTION_SUFFIX S_LINE_SUFFIX \
213 , ##__VA_ARGS__) : \
214 (void)NULL; \
215}

239#define LOG_BUFFER(level, buffer, size) \
240{ \
241 if(((uint8_t) level) <= ((uint8_t) DEBUG_LOG_MAX_LEVEL)) { \
242 for (uint16_t i = 0; i < size; i++) \
243 { \
244 Print_Log("%02X ", buffer[i]); \
245 if ((i & 0xF) == 0xF && i != (uint8_t)(size-1)) \
246 { \
247 Print_Log(S_LINEENDING); \
248 } \
249 } \
250 Print_Log(S_LINEENDING); \
251 } \
252}
